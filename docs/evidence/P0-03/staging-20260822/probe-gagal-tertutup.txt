// Demo TERKONTROL: dengan STORAGE_MODE=filesystem, vonis media lahir dari isi
// DISK LOKAL. MENULIS ke direktori sementara, membacanya lewat resolver, lalu
// menghapusnya. Nol jaringan, nol database, nol sentuhan ke staging.
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import crypto from "node:crypto";
const dir = fs.mkdtempSync(path.join(os.tmpdir(), "vonis-disk-"));
process.env.STORAGE_MODE = "filesystem";
process.env.STORAGE_DIR = dir;
const bytes = Buffer.from("BYTES-PALSU-DI-DISK-LOKAL");
fs.mkdirSync(path.join(dir, "p"), { recursive: true });
fs.writeFileSync(path.join(dir, "p/0.webp"), bytes);
fs.writeFileSync(path.join(dir, "p/0.webp.meta.json"), JSON.stringify({
  sha256: crypto.createHash("sha256").update(bytes).digest("hex"),
  jenis: "product_photo", layakReferensi: true,
  rasioAreaTeks: 0.004, jumlahKata: 2, alasan: "ditanam oleh probe", versiBukti: 1,
}));
const { resolveApprovedReference } = await import("./lib/product-truth");
const h = await resolveApprovedReference(["p/0.webp"]);
console.log("tersetujui          :", h.tersetujui.length);
console.log("ditolak             :", h.ditolak.length);
console.log("utama               :", h.utama ? h.utama.rel : "(nol)");
console.log("VONIS               :", h.utama ? "TERSETUJUI dari DISK LOKAL" : "ditolak");
fs.rmSync(dir, { recursive: true, force: true });
console.log("STORAGE_DIR dihapus :", !fs.existsSync(dir));
