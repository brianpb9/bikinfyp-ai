// Probe konfigurasi EFEKTIF. Tidak menyentuh jaringan, database, atau storage.
// Mencetak STATUS saja (NONEMPTY / EMPTY_OR_MISSING) — nilai tidak pernah dicetak.
const { config } = await import("./lib/config");
const { mediaStorage } = await import("./lib/storage");
const st = (v: unknown) => (typeof v === "string" && v.trim() !== "" ? "NONEMPTY" : "EMPTY_OR_MISSING");
console.log("R2_ENDPOINT          =", st(config.r2Endpoint));
console.log("R2_BUCKET            =", st(config.r2Bucket));
console.log("R2_ACCESS_KEY_ID     =", st(config.r2AccessKeyId));
console.log("R2_SECRET_ACCESS_KEY =", st(config.r2SecretAccessKey));
console.log("STORAGE_MODE         =", config.storageMode);
console.log("RACUN_DB_RUNTIME     =", config.dbRuntime);
try { mediaStorage(); console.log("mediaStorage()       = TIDAK_MELEMPAR (gagal TERBUKA)"); }
catch (e) { console.log("mediaStorage()       = MELEMPAR (gagal-tertutup)"); }
